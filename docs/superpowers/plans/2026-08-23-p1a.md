> **תיבות הסימון מתעדכנות תוך כדי.** המשימה הראשונה שאינה מסומנת היא נקודת ההמשך.

**Goal:** מטופל אומר איפה הוא גר ועד לאן ייסע, רואה רק מרפאות רלוונטיות, נותן
הסכמה מפורשת לפני שצילום הרנטגן שלו נשלח לאנשהו — ויכול למחוק הכל.

**Architecture:** מרחק הנכונות הוא תכונה של **הבקשה** ולא של המשתמש, כי אותו
אדם יטוס בשביל השתלה ולא בשביל סתימה. הכלל שממיר „מרחק נכונות” ל„אילו מדינות
להראות” הוא פונקציה טהורה, נפרדת מהמסד ומ-React, כדי שיהיה אפשר לבדוק אותו.
ההסכמה נשמרת בתבנית שכבר קיימת למרפאות: חותמת זמן פלוס **גרסת נוסח**.

**Tech Stack:** Next.js 16.2.11 (App Router), React 19.2, Prisma 7.8 + Postgres,
Clerk 7.5, Tailwind v4, vitest (node), `@vercel/blob`.

**Spec:** `docs/superpowers/specs/2026-08-23-p1-design.md` — פרקים 3, 4, 7.

```prisma
/// עד לאן המטופל מוכן לנסוע — עבור הבקשה הזו.

/// על הבקשה ולא על המשתמש: אותו אדם יטוס בשביל שתי השתלות ולא בשביל סתימה.
```

```ts
import type { TravelScope } from "@/generated/prisma/enums";

/**
 * אילו מדינות להראות למטופל, לפי מרחק הנכונות שהצהיר.
 *
 * טהורה במכוון — בלי מסד ובלי React — כי זה הכלל היחיד כאן שאפשר לטעות בו
 * בלי שאף מסך ייראה שבור.
 *
 * מחזירה `null` עבור ANY, ולא רשימה של כל המדינות. ההבדל נראה טכני והוא לא:
 * רשימה שנבנית עכשיו היא רשימה שהוקפאה, ומדינה שתופעל מחר לא תיכנס לשום
 * בקשה קיימת — בשקט. `null` אומר „בלי סינון”, וזה נשאר נכון לתמיד.
 */
export function destinationCountryCodes(
  scope: TravelScope,
  destinations: string[],
  patientCountry: string,
): string[] | null {
  if (scope === "ANY") return null;

  if (scope === "SELECTED") {
    const unique = [...new Set(destinations)];
    // רשימה ריקה תסנן הכל ותציג מסך ריק בלי הסבר. „בחרתי מדינות” בלי מדינות
    // הוא טופס שלא הושלם, לא בקשה להראות כלום.
    if (unique.length > 0) return unique;
  }

  return [patientCountry];
}
```

```ts
"use server";

import { revalidatePath } from "next/cache";
import { auth } from "@clerk/nextjs/server";
import { getDictionary } from "@/i18n/get-dictionary";
import { getRequestLocale } from "@/i18n/request-locale";
import { db } from "@/lib/db";

export type ActionResult = { ok: true } | { ok: false; error: string };

/**
 * איפה המטופל גר, ועד לאן ייסע עבור הבקשה הזו.
 *
 * המדינה נשמרת על המשתמש (היא תכונה שלו) ומרחק הנכונות על הבקשה (הוא תכונה
 * של הטיפול). שתיהן נכתבות יחד כי הן נשאלות באותו מסך.
 */
export async function saveTravelChoice(formData: FormData): Promise<ActionResult> {
  const e = (await getDictionary(await getRequestLocale())).errors;

  const { userId: clerkUserId } = await auth();
  if (!clerkUserId) return { ok: false, error: e.signInRequired };

  const user = await db.user.findUnique({ where: { clerkUserId }, select: { id: true } });
  if (!user) return { ok: false, error: e.userNotSynced };

  const requestId = String(formData.get("requestId") ?? "");
  const request = await db.request.findUnique({
    where: { id: requestId },
    select: { id: true, userId: true, status: true },
  });
  // אותה בקשה שאינה שלך ובקשה שאינה קיימת מקבלות אותה תשובה.
  if (!request || request.userId !== user.id) return { ok: false, error: e.requestNotFound };
  if (request.status !== "DRAFT") return { ok: false, error: e.requestLocked };

  const active = await db.country.findMany({
    where: { isActive: true },
    select: { code: true },
  });
  const activeCodes = new Set(active.map((c) => c.code));

  // מדינה שאינה פעילה קובעת מטבע שאין לו שער ומשטר פרטיות שלא נבדק. ערך
  // שהומצא בטופס לא ייכנס למסד.
  const countryCode = String(formData.get("countryCode") ?? "");
  if (!activeCodes.has(countryCode)) return { ok: false, error: e.mustPickCountry };

  const scopeRaw = String(formData.get("travelScope") ?? "");
  const travelScope =
    scopeRaw === "SELECTED" || scopeRaw === "ANY" || scopeRaw === "LOCAL" ? scopeRaw : "LOCAL";

  const destinations =
    travelScope === "SELECTED"
      ? [
          ...new Set(
            formData
              .getAll("destinations")
              .filter((v): v is string => typeof v === "string")
              .filter((code) => activeCodes.has(code)),
          ),
        ]
      : [];

  await db.$transaction([
    db.user.update({ where: { id: user.id }, data: { countryCode } }),
    db.request.update({
      where: { id: request.id },
      data: { travelScope, destinationCountries: destinations },
    }),
  ]);

  revalidatePath(`/request/${request.id}/dentists`);
  return { ok: true };
}
```

```ts
"use server";

import { revalidatePath } from "next/cache";
import { auth } from "@clerk/nextjs/server";
import { del } from "@vercel/blob";
import { getDictionary } from "@/i18n/get-dictionary";
import { getRequestLocale } from "@/i18n/request-locale";
import { db } from "@/lib/db";
import { audit } from "@/lib/audit";
import { logEvent } from "@/lib/log";

export type ActionResult = { ok: true } | { ok: false; error: string };

/**
 * מחיקת בקשה והקבצים שלה, לבקשת המטופל.
 *
 * הקבצים נמחקים **לפני** השורה. הסדר הזה נבחר: אם המחיקה מהאחסון נכשלת,
 * השורה נשארת ואפשר לנסות שוב — בעוד שהסדר ההפוך היה משאיר צילום רנטגן
 * בענן בלי שום רשומה שמצביעה עליו, כלומר קובץ שאיש לא יידע שצריך למחוק.
 */
export async function deleteRequest(requestId: string): Promise<ActionResult> {
  const e = (await getDictionary(await getRequestLocale())).errors;

  const { userId: clerkUserId } = await auth();
  if (!clerkUserId) return { ok: false, error: e.signInRequired };

  const user = await db.user.findUnique({
    where: { clerkUserId },
    select: { id: true, email: true },
  });
  if (!user) return { ok: false, error: e.userNotSynced };

  const request = await db.request.findUnique({
    where: { id: requestId },
    select: { id: true, userId: true, treatmentFileUrl: true, xrayFileUrl: true },
  });
  if (!request || request.userId !== user.id) return { ok: false, error: e.requestNotFound };

  for (const url of [request.treatmentFileUrl, request.xrayFileUrl]) {
    if (!url) continue;
    try {
      await del(url);
    } catch (err) {
      // קובץ שכבר אינו קיים אינו כישלון; כל שאר המקרים כן, והמחיקה נעצרת
      // כדי שהמטופל לא יקבל אישור על משהו שלא קרה.
      logEvent("error", "request.delete_blob_failed", { requestId, error: String(err) });
      return { ok: false, error: e.deleteFailed };
    }
  }

  // RequestDentist ו-Quote יורדים בקסקייד (onDelete: Cascade בסכימה).
  await db.request.delete({ where: { id: request.id } });

  await audit({
    actor: user.email,
    action: "request.deleted_by_patient",
    entity: "Request",
    entityId: requestId,
  });

  revalidatePath("/dashboard");
  return { ok: true };
}
```

הוסף `deleteFailed` לשני המילונים: עברית „מחיקת הקבצים נכשלה — נסו שוב”,
אנגלית „We couldn't delete the files — please try again”.
